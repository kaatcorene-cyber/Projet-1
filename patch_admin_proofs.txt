  const handleAddProof = async (e: any) => {
    e.preventDefault();
    if (!newProofImageUrl || !newProofTestimonial) return;
    setIsAddingProof(true);
    try {
      const text = `${newProofImageUrl} vient de retirer ${newProofTestimonial} FCFA avec succès !`;
      await supabase.from('proofs').insert({
        image_url: 'text_only',
        testimonial: text
      });
      setNewProofImageUrl("");
      setNewProofTestimonial("");
      fetchData(false);
    } catch (err) {
      console.error(err);
    } finally {
      setIsAddingProof(false);
    }
  };
